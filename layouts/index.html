{{- /* コンテンツの処理: content/ 直下にある quote-01.md などのファイルを取得し、 public/quote/001.json などの形式で出力するための準備をします。 */ -}} {{- $pages := where .Site.RegularPages "Section" "" -}} {{- $quotes := where $pages "File.BaseFileName" "like" "quote-" -}} {{- $sortedQuotes := sort $quotes "File.BaseFileName" -}} {{- $total := len $sortedQuotes -}} {{- range $index, $page := $sortedQuotes -}} {{- $num := add $index 1 -}} {{- $filename := printf "quote/%03d.json" $num -}} {{- $data := dict "title" $page.Title "author" ($page.Params.author | default "") "content" $page.Content -}} {{- $res := $data | jsonify | resources.FromString $filename -}} {{- $noop := $res.RelPermalink -}} {{- end -}} {{ .Title }}
