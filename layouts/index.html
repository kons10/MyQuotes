{{- /* content/ 直下の quote-01.md などを JSON 化し、quote/001.json として出力します。 */ -}} {{- $pages := where .Site.RegularPages "Section" "" -}} {{- $quotes := where $pages "File.BaseFileName" "like" "quote-" -}} {{- $sortedQuotes := sort $quotes "File.BaseFileName" -}} {{- $total := len $sortedQuotes -}} {{- range $index, $page := $sortedQuotes -}} {{- $num := add $index 1 -}} {{- $filename := printf "quote/%03d.json" $num -}} {{- $image := $page.Params.image | default "" -}} {{- if and $image (not (or (hasPrefix $image "https://") (hasPrefix $image "http://"))) -}} {{- $image = $image | relURL -}} {{- end -}} {{- $data := dict "title" $page.Title "author" ($page.Params.author | default "") "content" $page.Content "image" $image "imageAlt" ($page.Params.image_alt | default $page.Title) -}} {{- $res := $data | jsonify | resources.FromString $filename -}} {{- $noop := $res.RelPermalink -}} {{- end -}} {{ .Title }}
